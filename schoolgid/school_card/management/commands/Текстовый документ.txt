from django.core.management.base import BaseCommand
from django.apps import apps
import random

def generate_card_number():
    return str(random.randint(0, 99999999))  # 8-значный номер карты

class Command(BaseCommand):
    help = "Создаёт школьные карты для пользователей, у которых их нет"

    def handle(self, *args, **kwargs):
        User = apps.get_model('accounts', 'User')  # Получаем модель User через apps.get_model
        SchoolCard = apps.get_model('school_card', 'SchoolCard')  # Получаем модель SchoolCard

        users_without_cards = User.objects.filter(schoolcard=None)

        for user in users_without_cards:
            SchoolCard.objects.create(
                user=user,
                balance=0.00,
                card_number=generate_card_number()
            )
            self.stdout.write(self.style.SUCCESS(f"Добавлена карта для {user.username}"))

        self.stdout.write(self.style.SUCCESS("Все пользователи теперь имеют карты!"))
